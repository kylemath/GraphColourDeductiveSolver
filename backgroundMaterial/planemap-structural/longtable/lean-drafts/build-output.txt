      SimpleGraph.VacancySlide.ProperOff (SimpleGraph.BeltDraft.G n) (SimpleGraph.BeltVacancyTeamA.Vertex.u 0) c →

                SimpleGraph.BeltDraft.Filled t ∧
                  SimpleGraph.VacancySlide.ProperOff (SimpleGraph.BeltDraft.G n) t.1 t.2 ∧
                    SimpleGraph.BeltDraft.IsBelt t.1 ∧
                      t.2 SimpleGraph.BeltVacancyTeamA.Vertex.a = c SimpleGraph.BeltVacancyTeamA.Vertex.a ∧
                        t.2 SimpleGraph.BeltVacancyTeamA.Vertex.b = c SimpleGraph.BeltVacancyTeamA.Vertex.b
