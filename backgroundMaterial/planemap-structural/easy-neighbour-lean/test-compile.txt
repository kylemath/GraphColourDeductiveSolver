def SimpleGraph.VacancyEasyNeighbour.OneSwapAt.{u_1, u_2} : {V : Type u_1} →
  {C : Type u_2} → SimpleGraph V → [DecidableEq C] → V → (V → C) → Prop :=
fun {V} {C} G [DecidableEq C] h c => Target G h c ∨ ∃ d, KempeStep G h c d ∧ Target G h d
def SimpleGraph.VacancyEasyNeighbour.EasyAt.{u_1, u_2} : {V : Type u_1} →
  {C : Type u_2} → SimpleGraph V → [DecidableEq C] → V → Prop :=
fun {V} {C} G [DecidableEq C] u => ∀ (c : V → C), ProperOff G u c → OneSwapAt G u c
def SimpleGraph.VacancyEasyNeighbour.SlideAccess.{u_1, u_2} : {V : Type u_1} →
  {C : Type u_2} → SimpleGraph V → [DecidableEq C] → V → V → (V → C) → Prop :=
fun {V} {C} G [DecidableEq C] h u c => UniqueAt G h u c ∨ ∃ d, KempeStep G h c d ∧ UniqueAt G h u d
